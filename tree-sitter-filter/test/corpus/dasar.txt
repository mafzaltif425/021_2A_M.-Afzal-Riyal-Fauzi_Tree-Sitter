==================
2. Prioritas and lebih tinggi dari or
==================

a = 1 or b = 2 and c = 3

---

(query
  (or_expression
    left: (comparison
      field: (identifier)
      value: (number))
    right: (and_expression
      left: (comparison
        field: (identifier)
        value: (number))
      right: (comparison
        field: (identifier)
        value: (number)))))

==================
3. Pengelompokan dengan tanda kurung
==================

(a = 1 or b = 2) and c = 3

---

(query
  (and_expression
    left: (parenthesized
      (or_expression
        left: (comparison
          field: (identifier)
          value: (number))
        right: (comparison
          field: (identifier)
          value: (number))))
    right: (comparison
      field: (identifier)
      value: (number))))

6. Operator contains untuk pencarian teks

(query

==================
7. Query kompleks gabungan admin enterprise
==================

(status = "active" or role in ("admin", "supervisor")) and not department.is_locked = true and title contains "Manager"

---

(query
  (and_expression
    left: (and_expression
      left: (parenthesized
        (or_expression
          left: (comparison
            field: (identifier)
            value: (string))
          right: (in_expression
            field: (identifier)
            values: (string)
            values: (string))))
      right: (not_expression
        operand: (comparison
          field: (identifier)
          value: (boolean))))
    right: (contains_expression
      field: (identifier)
      value: (string))))
